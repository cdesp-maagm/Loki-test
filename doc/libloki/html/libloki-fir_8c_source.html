50static inline void LokiFIRFilter_allocate_samples(LokiFIRFilter *self) {
51 size_t buffer_length = self->length;

60LokiFIRFilter *LokiFIRFilter_new(const float *coeffs, size_t length, size_t decimation) {
61 LokiFIRFilter * self = calloc(1, sizeof(LokiFIRFilter));
62 if (!self) {
63 return NULL;
64 }
65
66 // Ajuster la longueur du filtre pour qu'elle soit multiple de decimation.
67 self->phase_length = length / decimation;
68 if (length % decimation) {
69 self->phase_length ++;
70 }
71 #ifndef NO_SIMD
72 // Si la vectorisation est activée, la longueur de chaque sous-filtre sera
73 // ajustée au multiple de 4 immédiatement supérieur.
74 if (self->phase_length % 4) {
75 self->phase_length += 4 - self->phase_length % 4;
76 }
77 #endif
78
79 self->length = self->phase_length * decimation;
80 self->decimation = decimation;
81
82 if (length != self->length) {
83 fprintf(stderr, "LokiFIRFilter_new: Filter length was expanded from %zu to %zu\n", length, self->length);
84 }
85
86 // Allouer les tableaux des coefficients et des échantillons précédents.
87 self->coeffs = self->coeff_ptr = calloc(self->length, sizeof(float));
88 self->coeffs_end = self->coeffs + self->length;
89 LokiFIRFilter_allocate_samples(self);
90 if (!self->coeffs || !self->samples) {
91 LokiFIRFilter_del(self);
92 return NULL;
93 }
94
95 // Ranger les coefficients en les répartissant par phase
96 // et en renversant l'ordre des coefficients.
97 for (size_t i = 0; i < length; i ++) {
98 size_t p = i % decimation; // Numéro de phase (< decimation)
99 size_t q = (i - p) / decimation; // Indice dans la phase (< n)
100 self->coeffs[p * self->phase_length + q] = coeffs[length - i - 1];
101 }
102
103 return self;
104}

125static float bessel0(float x) {
126 float d = 0, ds = 1, s = 1;
127 do {
128 d += 2;
129 ds *= x * x / d / d;
130 s += ds;
131 }
132 while (ds > s * 1e-6f);
133 return s;
134}

136LokiFIRFilter *LokiFIRFilter_new_kaiser_bessel(float fs, float fa, float fb, float att, size_t length, size_t decimation) {
137 // Ajuster la taille du filtre à un nombre impair de coefficients.
138 const size_t half_length = (length - 1) / 2;
139 length = half_length * 2 + 1;
140
141 // Allouer le tableau des coefficients.
142 float *coeffs = calloc(length, sizeof(float));
143 if (!coeffs) {
144 return NULL;
145 }
146
147 // Calculer la partie droite de la réponse impulsionnelle du fitre idéal.
148 coeffs[half_length] = (fb - fa) * 2 / fs;
149 for (size_t i = 1; i <= half_length; i ++) {
150 coeffs[half_length + i] = (sinf(M_PI * 2 * i * fb / fs) - sinf(M_PI * 2 * i * fa / fs)) / i / M_PI;
151 }
152
153 // Calculer le facteur de forme de la fenêtre de Kaiser-Bessel.
154 float alpha = att < 21 ? 0.0f :
155 att > 50 ? 0.1102f * (att - 8.7f) :
156 0.5842f * powf((att - 21.0f), 0.4f) + 0.07886f * (att - 21.0f);
157
158 // Appliquer la fenêtre sur la réponse idéale.
159 float alpha0 = bessel0(alpha);
160 for (size_t i = 0; i <= half_length; i ++) {
161 coeffs[half_length + i] *= bessel0(alpha * sqrtf(1.0f - (float)(i * i) / half_length / half_length)) / alpha0;
162 coeffs[half_length - i] = coeffs[half_length + i];
163 }
164
165 LokiFIRFilter *self = LokiFIRFilter_new(coeffs, length, decimation);
166 free(coeffs);
167
168 return self;
169}

175bool LokiFIRFilter_work(LokiFIRFilter *self, float in, float *out) {
176 // Copie locale de l'état du filtre.
177 const size_t phase_length = self->phase_length;
178 const size_t decimation = self->decimation;
179 float *phase_ptr = self->phase_ptr;
180 float *const phase_end = phase_ptr + phase_length;
181 size_t sample_index = self->sample_index;
182 float *sample_ptr = phase_ptr + sample_index;
183 float *const samples = self->samples;
184 float *const samples_end = self->samples_end;
185 const float *coeff_ptr = self->coeff_ptr;
186 const float *const coeffs = self->coeffs;
187 const float *const coeffs_end = self->coeffs_end;
188 float acc = self->acc;
189
190 #ifdef NO_SIMD
191 // Version de référence sans vectorisation
192
193 // Insérer le nouvel échantillon dans le tampon de réception.
194 *sample_ptr++ = in;
195
196 for (size_t i = 0; i < phase_length; i ++) {
197 if (sample_ptr == phase_end) {
198 sample_ptr = phase_ptr;
199 }
200
201 acc += (*sample_ptr ++) * (*coeff_ptr ++);
202
203 if (coeff_ptr == coeffs_end) {
204 coeff_ptr = coeffs;
205 }
206 }
207 #else
208 // Vectorisation ARM NEON
209 float32x4_t vsamp, vcoeff;
210 float32x4_t vacc = vmovq_n_f32(0);
211
212 // Insérer le nouvel échantillon dans le tampon de réception.
213 // Les trois premiers échantillons du buffer sont dupliqués en fin
214 // de buffer de la phase courante. Cela permet de traiter le buffer
215 // par vecteurs de 4 échantillons même si sample_ptr n'est pas aligné
216 // sur un multiple de 4.
217 if (sample_index < 3) {
218 sample_ptr[phase_length] = in;
219 }
220 *sample_ptr++ = in;
221
222 float *saved_sample_ptr = sample_ptr;
223
224 // Traiter les échantillons par vecteurs de 4 jusqu'à la fin du buffer
225 // pour la phase courante.
226 while (sample_ptr < phase_end) {
227 vsamp = vld1q_f32(sample_ptr);
228 vcoeff = vld1q_f32(coeff_ptr);
229 vacc = vmlaq_f32(vacc, vsamp, vcoeff);
230
231 sample_ptr += 4;
232 coeff_ptr += 4;
233
234 if (coeff_ptr == coeffs_end) {
235 coeff_ptr = coeffs;
236 }
237 }
238
239 // Lorsque la fin du buffer est dépassée, reprendre au début.
240 sample_ptr -= phase_length;
241
242 // Traiter les échantillons à partir du début du buffer
243 // jusqu'à l'emplacement de départ.
244 while (sample_ptr != saved_sample_ptr) {
245 vsamp = vld1q_f32(sample_ptr);
246 vcoeff = vld1q_f32(coeff_ptr);
247 vacc = vmlaq_f32(vacc, vsamp, vcoeff);
248
249 sample_ptr += 4;
250 coeff_ptr += 4;
251
252 if (coeff_ptr == coeffs_end) {
253 coeff_ptr = coeffs;
254 }
255 }
256
257 // Calculer la somme des quatre accumulateurs.
258 float32x2_t vacc2 = vadd_f32(vget_low_f32(vacc), vget_high_f32(vacc));
259 acc += vget_lane_f32(vacc2, 0) + vget_lane_f32(vacc2, 1);
260
261 phase_ptr += 3;
262 #endif
263
264 // Passer à la phase suivante dans le tableau des échantillons.
265 phase_ptr += phase_length;
266 if (phase_ptr == samples_end) {
267 phase_ptr = samples;
268 sample_index ++;
269 if (sample_index == phase_length) {
270 sample_index = 0;
271 }
272 }
273
274 // Lorsqu'on revient au premier coefficient,
275 // le calcul d'un échantillon de sortie est terminé.
276 bool res = coeff_ptr == coeffs;
277 if (res) {
278 *out = acc;
279 self->acc = 0;
280 }
281 else {
282 self->acc = acc;
283 }
284
285 // Mise à jour de l'état du filtre.
286 self->phase_ptr = phase_ptr;
287 self->sample_index = sample_index;
288 self->coeff_ptr = (float*)coeff_ptr;
289
290 return res;
291}
