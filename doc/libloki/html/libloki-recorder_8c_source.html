98LokiRecorder *LokiRecorder_new(const char *filename, const char *filetype, size_t max_channels, int32_t channels, size_t sample_size, float sampling_frequency, float duration, float duration_per_file, void *config) {
99 LokiRecorder *self = calloc(1, sizeof(LokiRecorder));
100 if (!self) {
101 return NULL;
102 }
103
104 self->filename = calloc(strlen(filename) + strlen(filetype) + FILENAME_SUFFIX_LEN + 1, 1);
105 if (!self->filename) {
106 LokiRecorder_del(self);
107 return NULL;
108 }
109 strcpy(self->filename, filename);
110 self->filetype = strdup(filetype);
111
112 // Compter le nombre de voies activées dans le masque channels.
113 self->channel_count = 0;
114 for (size_t c = 0; c < max_channels; c ++) {
115 if (channels & (1 << c)) {
116 self->channel_count ++;
117 }
118 }

120 self->max_channels = max_channels;
121 self->channels = channels;
122 self->sampling_frequency = sampling_frequency;
123 self->sample_size = sample_size;
124 self->sample_count = self->channel_count * ceilf(duration * sampling_frequency);
125 self->multifile = duration_per_file > 0;
126 self->sample_count_per_file = self->multifile ?
127 self->channel_count * ceilf(duration_per_file * sampling_frequency) :
128 self->sample_count;
129
130 fprintf(stderr, "LokiRecorder_new (%s): will record %zu samples (%zu samples per file)\n", self->filename, self->sample_count, self->sample_count_per_file);
131
132 LokiRecorder_set_formatter(self, config);
133
134 pthread_mutex_init(&self->mutex, NULL);
135 pthread_cond_init(&self->recording_done, NULL);
136
137 return self;
138}

161static void *LokiRecorder_work(void *ptr) {
162 LokiRecorder *self = ptr;
163
164 pthread_setname_np(self->thread, "LokiRecorder");
165
166 const size_t in_length = LokiPacket_get_length(self->in, self->sample_size);
167
168 uint64_t sample_index = 0;
169 size_t sample_index_per_file = 0;
170
171 // Obtenir un pointeur vers la date dans le nom de fichier.
172 char *filename_suffix_date = self->filename + strlen(self->filename);
173 char *filename_suffix_msec = filename_suffix_date + FILENAME_SUFFIX_DATE_LEN;
174
175 // Ce booléen évite de terminer sans avoir écrit un seul paquet.
176 bool data_written = false;
177
178 fprintf(stderr, "LokiRecorder_work (%s): starting.\n", self->filename);
179
180 while (!self->stop_request) {
181 LokiPacket *in_packet = LokiFIFO_get_consumer_item(self->in, self->cid);
182 if (!in_packet) {
183 break;
184 }
185 void *in_ptr = in_packet->buffer;
186
187 bool file_change = sample_index_per_file == 0;
188
189 // Si l'échantillon courant est le premier d'un fichier,
190 // construire un nouveau nom de fichier.
191 if (file_change) {
192 time_t time_sec = LokiTime_seconds(in_packet->timestamp);
193 time_t time_msec = LokiTime_milliseconds(in_packet->timestamp);
194 struct tm file_time;
195 gmtime_r(&time_sec, &file_time);
196 strftime(filename_suffix_date, FILENAME_SUFFIX_DATE_LEN + 1, "_%Y%m%dT%H%M%S", &file_time);
197 sprintf(filename_suffix_msec, ".%03luZ.%s", time_msec, self->filetype);
198 }
199
200 // Écrire le paquet courant dans le fichier de destination.
201 LokiFormatter_work(self->formatter, self->filename, in_packet->timestamp, file_change, in_ptr, in_length, self->max_channels, self->channels, self->sample_size, self->sampling_frequency);
202 data_written = true;
203
204 // Mettre à jour le compteur d'échantillons pour le fichier courant.
205 // Si le fichier est plein, remettre le compteur à zéro et fermer le fichier.
206 sample_index_per_file += in_length;
207 if (self->multifile && sample_index_per_file > self->sample_count_per_file) {
208 sample_index_per_file = 0;
209 }
210
211 LokiFIFO_pull(self->in, self->cid);
212
213 // Mettre à jour le compteur d'échantillons global.
214 // Si la durée d'acquisition est atteinte, terminer.
215 sample_index += in_length;
216 if (data_written && self->sample_count > 0 && sample_index >= self->sample_count) {
217 break;
218 }
219 }
220
221 fprintf(stderr, "LokiRecorder_work (%s): done.\n", self->filename);
222
223 LokiFormatter_close(self->formatter);
224
225 pthread_mutex_lock(&self->mutex);
226 self->is_recording = false;
227 pthread_cond_signal(&self->recording_done);
228 pthread_mutex_unlock(&self->mutex);
229}

262void LokiRecorder_wait(LokiRecorder *self) {
263 LokiRecorder_waiting_instance = self;
264 signal(SIGKILL, LokiRecorder_signal_handler);
265 signal(SIGTERM, LokiRecorder_signal_handler);
266 signal(SIGINT, LokiRecorder_signal_handler);
267
268 pthread_mutex_lock(&self->mutex);
269 if (self->is_recording) {
270 pthread_cond_wait(&self->recording_done, &self->mutex);
271 }
272 pthread_mutex_unlock(&self->mutex);
