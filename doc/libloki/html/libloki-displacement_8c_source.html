43LokiDisplacement *LokiDisplacement_new(LokiEllipseEstimator *estimator, LokiFIRFilter *filter, unsigned throttle, float scale, float tau[static 3], float sampling_frequency) {
44 LokiDisplacement *self = calloc(1, sizeof(LokiDisplacement));
45 if (!self) {
46 return NULL;
47 }
48
49 self->estimator_throttle = throttle;
50 self->estimator_cnt = 0;
51 self->scale = scale;
52 self->estimator = estimator;
53 self->filter = filter;
54
55 for (size_t i = 0; i < 3; i ++) {
56 const float ts = tau[i] * sampling_frequency;

58 self->iir_b[i] = 1.0f / (ts + 1);
59 }

61 return self;

82bool LokiDisplacement_work(LokiDisplacement *self, float x, float y, float *x_norm, float *y_norm, float *out) {
83 // Calcul des paramètres de l'ellipse et de l'angle courant.
84 self->estimator_cnt ++;
85 if (self->estimator_cnt >= self->estimator_throttle) {
86 LokiEllipseEstimator_work(self->estimator, x, y);
87 self->estimator_cnt = 0;
88 }
89
90 LokiEllipse ellipse;
91 LokiEllipse_set_from_coeffs(&ellipse, LokiEllipseEstimator_get_state(self->estimator));
92
93 // Éviter les sauts entre -pi/2 et +pi/2 lorsque l'ellipse est presque verticale.
94 float dalpha = ellipse.alpha - self->alpha_prev;
95 self->alpha_prev = ellipse.alpha;
96
97 if (dalpha > M_PI_2 || dalpha < -M_PI_2) {
98 self->alpha_ref = M_PI - self->alpha_ref;
99 }
100
101 ellipse.alpha += self->alpha_ref;
102
103 // Lisser les paramètres de l'ellipse.
104 const float *const a = self->iir_a;
105 const float *const b = self->iir_b;
106 self->ellipse.A = a[0] * self->ellipse.A + b[0] * ellipse.A;
107 self->ellipse.B = a[0] * self->ellipse.B + b[0] * ellipse.B;
108 self->ellipse.xc = a[1] * self->ellipse.xc + b[1] * ellipse.xc;
109 self->ellipse.yc = a[1] * self->ellipse.yc + b[1] * ellipse.yc;
110 self->ellipse.alpha = a[2] * self->ellipse.alpha + b[2] * ellipse.alpha;
111
112 // Calculer la phase courante.
113 float phi = LokiEllipse_get_angle(&self->ellipse, x, y, x_norm, y_norm);
114
115 // Dérouler la phase pour éviter les discontinuités au passage de -pi à +pi.
116 float dphi = phi - self->phi_prev;
117 self->phi_prev = phi;
118
119 if (dphi > M_PI) {
120 self->phi_turns --;
121 }
122 else if (dphi < -M_PI) {
123 self->phi_turns ++;
124 }
125
126 phi += M_PI * 2 * self->phi_turns;
127
128 // Filtrage passe-bas pour éliminer le déplacement fictif.
129 if (LokiFIRFilter_work(self->filter, phi, &phi)) {
130 // Mémoriser la dernière valeur filtrée, pour utilisation comme
131 // référence dans LokiDisplacement_reset.
132 self->phi_filt = phi;
133 // Calculer le déplacement et affecter l'échantillon de destination.
134 *out = (phi - self->phi_ref) * self->scale;
135 return true;
136 }
137
138 return false;
139}
