10void LokiEllipse_set_from_coeffs(LokiEllipse *self, const LokiEllipseEquation coeffs) {
11 float a = coeffs[0];
12 float b = coeffs[1];
13 float c = 1.0f - a;
14 float d = coeffs[2];
15 float e = coeffs[3];
16 float f = coeffs[4];
17
18 float ac = a * c;
19 float ae = a * e;
20 float cma = c - a;
21 float b2 = b * b;
22 float bd = b * d;
23 float be = b * e;
24 float cd = c * d;
25
26 float b2m4ac = b2 - ac * 4;
27 float samcpb2 = sqrtf(cma * cma + b2) / 2;
28 float A0 = 0.5f + samcpb2; // 0.5 == (a + c) / 2
29 float B0 = 0.5f - samcpb2; // 0.5 == (a + c) / 2
30 float xc = (cd * 2 - be) / b2m4ac;
31 float yc = (ae * 2 - bd) / b2m4ac;
32 float df = a * xc * xc + b * xc * yc + c * yc * yc - f;
33
34 // FIXME Est-il utile de calculer alpha sachant que par la suite, on recalcule son cos et son sin?
35 self->alpha = atan2f(-b, cma) / 2;
36 self->xc = xc;
37 self->yc = yc;
38 self->A = sqrtf(fabsf(df / B0));
39 self->B = sqrtf(fabsf(df / A0));
40}

42void LokiEllipse_get_coeffs(const LokiEllipse *self, LokiEllipseEquation coeffs) {
43 float A2 = self->A * self->A;
44 float B2 = self->B * self->B;
45 float cos_alpha = cosf(self->alpha);
46 float sin_alpha = sinf(self->alpha);
47 float cos_alpha2 = cos_alpha * cos_alpha;
48 float sin_alpha2 = sin_alpha * sin_alpha;
49 float xc2 = self->xc * self->xc;
50 float yc2 = self->yc * self->yc;
51 float K = A2 + B2;
52
53 coeffs[0] = (A2 * sin_alpha2 + B2 * cos_alpha2) / K;
54 coeffs[1] = (B2 - A2) * sin_alpha * cos_alpha * 2 / K;
55 coeffs[2] = -coeffs[0] * self->xc * 2 - coeffs[1] * self->yc;
56 coeffs[3] = (coeffs[0] - 1) * self->yc * 2 - coeffs[1] * self->xc;
57 coeffs[4] = coeffs[0] * xc2 + coeffs[1] * self->xc * self->yc + (1.0f - coeffs[0]) * yc2 - A2 * B2 / K;
58}

60void LokiEllipse_get_point(const LokiEllipse *self, float theta, float *x, float *y) {
61 complex float z = (self->A * cosf(theta)) + I * (self->B * sinf(theta));
62 z *= cosf(self->alpha) + I * sinf(self->alpha);
63
64 *x = self->xc + crealf(z);
65 *y = self->yc + cimagf(z);
66}

68float LokiEllipse_get_angle(const LokiEllipse *self, float x, float y, float *x_norm, float *y_norm) {
69 complex float z = x + I * y;
70 // Translation pour recentrer l'ellipse autour de (0, 0).
71 z -= self->xc + I * self->yc;
72 // Rotation pour aligner le grand axe sur l'axe horizontal.
73 z *= cosf(self->alpha) - I * sinf(self->alpha);
74 // Normalisation pour transformer l'ellipse en un cercle unité.
75 *x_norm = x = crealf(z) / self->A;
76 *y_norm = y = cimagf(z) / self->B;
77 // Calcul de l'angle.
78 return atan2f(y, x);
79}
