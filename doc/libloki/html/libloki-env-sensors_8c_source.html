174static int32_t LokiEnvSensors_bmp390_compensate_temperature(LokiEnvSensors *self, uint32_t data) {
175 uint32_t nvm_reg0 = LokiEnvSensors_get_reg(self, LOKI_ENV_SENSORS_REG_BMP390_NVM0);
176 uint32_t nvm_reg1 = LokiEnvSensors_get_reg(self, LOKI_ENV_SENSORS_REG_BMP390_NVM1);
177
178 uint16_t t1 = nvm_reg0 >> 16;
179 uint16_t t2 = nvm_reg0;
180 int8_t t3 = nvm_reg1;
181
182 uint64_t partial_data1 = (uint64_t)data - (uint64_t)t1 * 256;
183 int64_t partial_data2 = t2 * partial_data1;
184 int64_t partial_data3 = partial_data1 * partial_data1;
185 int64_t partial_data4 = partial_data3 * t3;
186 int64_t partial_data5 = partial_data2 * 262144 + partial_data4;
187 int64_t partial_data6 = partial_data5 / 4294967296U;
188 self->bmp390_t_fine = partial_data6;
189 return partial_data6 * 25 / 16384;
190}

193static int32_t LokiEnvSensors_bmp390_compensate_pressure(const LokiEnvSensors *self, uint32_t data) {
194 uint32_t nvm_reg2 = LokiEnvSensors_get_reg(self, LOKI_ENV_SENSORS_REG_BMP390_NVM2);
195 uint32_t nvm_reg3 = LokiEnvSensors_get_reg(self, LOKI_ENV_SENSORS_REG_BMP390_NVM3);
196 uint32_t nvm_reg4 = LokiEnvSensors_get_reg(self, LOKI_ENV_SENSORS_REG_BMP390_NVM4);
197 uint32_t nvm_reg5 = LokiEnvSensors_get_reg(self, LOKI_ENV_SENSORS_REG_BMP390_NVM5);
198
199 int16_t p1 = nvm_reg2 >> 16;
200 int16_t p2 = nvm_reg2;
201 int8_t p3 = nvm_reg3 >> 24;
202 int8_t p4 = nvm_reg3 >> 16;
203 uint16_t p5 = nvm_reg3;
204 uint16_t p6 = nvm_reg4 >> 16;
205 int8_t p7 = nvm_reg4 >> 8;
206 int8_t p8 = nvm_reg4;
207 int16_t p9 = nvm_reg5 >> 16;
208 int8_t p10 = nvm_reg5 >> 8;
209 int8_t p11 = nvm_reg5;
210
211 int64_t t_fine = self->bmp390_t_fine;
212 int64_t partial_data1 = t_fine * t_fine;
213 int64_t partial_data2 = partial_data1 / 64;
214 int64_t partial_data3 = partial_data2 * t_fine / 256;
215 int64_t partial_data4 = p8 * partial_data3 / 32;
216 int64_t partial_data5 = p7 * partial_data1 * 16;
217 int64_t partial_data6 = p6 * t_fine * 4194304;
218 int64_t offset = (int64_t)p5 * (int64_t)140737488355328U + partial_data4 + partial_data5 + partial_data6;
219 partial_data2 = (int64_t)p4 * partial_data3 / 32;
220 partial_data4 = p3 * partial_data1 * 4;
221 partial_data5 = ((int64_t)p2 - 16384) * t_fine * 2097152;
222 int64_t sensitivity = ((int64_t)p1 - 16384) * (int64_t)70368744177664U + partial_data2 + partial_data4 + partial_data5;
223 partial_data1 = sensitivity / 16777216 * data;
224 partial_data2 = (int64_t)p10 * t_fine;
225 partial_data3 = partial_data2 + 65536 * (int64_t)p9;
226 partial_data4 = partial_data3 * data / 8192;
227 partial_data5 = partial_data4 * data / 512;
228 partial_data6 = (uint64_t)data * (uint64_t)data;
229 partial_data2 = (int64_t)p11 * partial_data6 / 65536;
230 partial_data3 = partial_data2 * data / 128;
231 partial_data4 = offset / 4 + partial_data1 + partial_data5 + partial_data3;
232 return (uint64_t)partial_data4 * 25 / 1099511627776U;
233}

235LokiEnvSensors *LokiEnvSensors_new(void) {
236 LokiEnvSensors *self = calloc(1, sizeof(LokiEnvSensors));
237 if (!self) {
238 return NULL;
239 }
240
241 self->file = fopen("/dev/loki-env-sensors", "r");
242 if (!self->file) {
243 LokiEnvSensors_del(self);
244 return NULL;
245 }
246
247 self->fileno = fileno(self->file);
248
249 self->clk_frequency = LokiEnvSensors_get_reg(self, LOKI_ENV_SENSORS_REG_CLK_FREQUENCY);
250 self->channels = LokiEnvSensors_get_reg(self, LOKI_ENV_SENSORS_REG_CHANNELS);
251 self->channel_output_width = LokiEnvSensors_get_reg(self, LOKI_ENV_SENSORS_REG_CHANNEL_OUTPUT_WIDTH);
252 self->data_width = LokiEnvSensors_get_reg(self, LOKI_ENV_SENSORS_REG_DATA_WIDTH);
253 self->cycles_per_sample_max = LokiEnvSensors_get_reg(self, LOKI_ENV_SENSORS_REG_CYCLES_PER_SAMPLE_MAX);
254 self->channel_mask = (1 << self->channel_output_width) - 1;
255 self->data_mask = (1 << self->data_width) - 1;
256 self->data_sign_mask = 1 << (self->data_width - 1);
257
258 // Allouer un tableau pour les facteurs de mise à l'échelle,
259 // et les initialiser à 1.
260 self->scales = calloc(self->channels, sizeof(float));
261 if (!self->scales) {
262 LokiEnvSensors_del(self);
263 return NULL;
264 }
265
266 for (size_t i = 0; i < self->channels; i ++) {
267 self->scales[i] = 1.0f;
268 }
269
270 // Le nombre d'octets d'un transfert DMA doit être multiple de
271 // * la taille d'une page mémoire
272 // * la taille d'un échantillon
273 uint32_t page_size = getpagesize();
274 uint32_t a = page_size;
275 uint32_t b = sizeof(LokiDMASample);
276 while (a != b) {
277 if (a > b) {
278 a -= b;
279 }
280 if (b > a) {
281 b -= a;
282 }
283 }
284
285 LokiEnvSensors_set_reg(self, LOKI_ENV_SENSORS_REG_PACKET_LENGTH, page_size / a);
286
287 return self;
288}

344float LokiEnvSensors_get_last_data(const LokiEnvSensors *self, size_t channel) {
345 int32_t value;
346 if (channel < self->channels - 2) {
347 value = LokiEnvSensors_get_reg(self, LOKI_ENV_SENSORS_REG_LAST_DATA + channel);
348 }
349 else {
350 uint32_t t_raw = LokiEnvSensors_get_reg(self, LOKI_ENV_SENSORS_REG_LAST_DATA + self->channels - 2);
351 value = LokiEnvSensors_bmp390_compensate_temperature(self, t_raw);
352 if (channel == self->channels - 1) {
353 uint32_t p_raw = LokiEnvSensors_get_reg(self, LOKI_ENV_SENSORS_REG_LAST_DATA + channel);
354 value = LokiEnvSensors_bmp390_compensate_pressure(self, p_raw);
355 }
356 }
357 return self->scales[channel] * value;
358}

382static void LokiEnvSensors_set_packet_item(const LokiEnvSensors *self, LokiEnvSensorsSample *item, LokiDMASample raw_item) {
383 // Lire la valeur de l'échantillon, alignée à droite en complément à deux.
384 int32_t data_raw = raw_item & self->data_mask;
385 raw_item >>= self->data_width;
386
387 // Lire le numéro de la voie.
388 item->channel = raw_item & self->channel_mask;
389
390 // BMP390 : effectuer la compensation des mesures de température et de pression.
391 // Autres voies: effectuer l'extension du bit de signe.
392 if (item->channel == self->channels - 2) {
393 data_raw = LokiEnvSensors_bmp390_compensate_temperature(self, data_raw);
394 }
395 else if (item->channel == self->channels - 1) {
396 data_raw = LokiEnvSensors_bmp390_compensate_pressure(self, data_raw);
397 }
398 else if (data_raw & self->data_sign_mask) {
399 data_raw |= ~self->data_mask;
400 }
401
402 // Convertir les coordonnées en flottants si nécessaire.
403 if (self->raw) {
404 item->data_raw = data_raw;
405 }
406 else {
407 item->data = self->scales[item->channel] * data_raw;
408 }
409}

433static void *LokiEnvSensors_work(void *ptr) {
434 LokiEnvSensors *self = ptr;
435
436 pthread_setname_np(self->thread, "LokiEnvSensors");
437
438 // Obtenir un pointeur vers le buffer courant des échantillons mise en forme.
439 const size_t out_length = LokiPacket_get_length(self->out, sizeof(LokiEnvSensorsSample));
440 LokiPacket *out_packet = LokiFIFO_get_producer_item(self->out);
441 if (!out_packet) {
442 return NULL;
443 }
444 LokiEnvSensorsSample *out_ptr = (LokiEnvSensorsSample*)out_packet->buffer;
445 LokiEnvSensorsSample *out_end = out_ptr + out_length;
446
447 // Avec la fonction fread, allouer un buffer de données.
448 // Avec l'accès par mmap, le buffer est fourni par le pilote.
449 const size_t in_length = LokiEnvSensors_get_packet_length(self);
450 #ifndef USE_MMAP
451 LokiDMASample *in_buffer = calloc(in_length, sizeof(LokiDMASample));
452 LokiDMASample *in_end = in_buffer + in_length;
453 #else
454 size_t in_length_bytes = in_length * sizeof(LokiDMASample);
455 #endif
456
457 // Calculer la durée moyenne d'un paquet d'entrée, en nanosecondes.
458 double in_period_ns = 1.0e9 * in_length / self->channels / LokiEnvSensors_get_sampling_frequency(self);
459
460 // Calculer le modulo du compteur de paquets du FPGA.
461 uint32_t packet_count_mod = 1 << (sizeof(LokiDMASample) * 8 - self->channel_output_width - self->data_width);
462
463 bool first_packet = true;
464 uint32_t packet_index_prev = 0;
465 LokiTimestamp in_timestamp;
466
467 // Démarrer l'acquisition.
468 LokiEnvSensors_dma_start(self);
469
470 while (!self->stop_request) {
471 // Lire un buffer d'échantillons en provenance du DMA.
472 #ifndef USE_MMAP
473 if (fread(in_buffer, sizeof(LokiDMASample), in_length, self->file) != in_length) {
474 fprintf(stderr, "LokiEnvSensors_work: read failed (error: %d, %s)\n", errno, strerror(errno));
475 break;
476 }
477 #else
478 // Accès au buffer DMA en mode zero-copy.
479 // Contrairement à l'acquisition des voies optiques, l'appel à mmap
480 // peut bloquer le processus pendant une longue durée.
481 // On utilise le timeout pour le débloquer jusqu'au succès.
482 LokiDMASample *in_buffer;
483 do {
484 in_buffer = mmap(NULL, in_length_bytes, PROT_READ, MAP_PRIVATE, self->fileno, 0);
485 } while (in_buffer == MAP_FAILED && errno == ETIMEDOUT);
486
487 if (in_buffer == MAP_FAILED) {
488 fprintf(stderr, "LokiEnvSensors_work: mmap failed (error: %d, %s)\n", errno, strerror(errno));
489 break;
490 }
491
492 LokiDMASample *in_end = in_buffer + in_length;
493 #endif
494
495 // Attendre le paquet numéro 0 pour commencer.
496 uint32_t packet_index = LokiEnvSensors_get_packet_index(self, *in_buffer);
497
498 // Prendre pour référence temporelle la date du premier paquet.
499 // Les dates des autres paquets seront calculées en fonction du nombre d'échantillons.
500 if (first_packet && !packet_index) {
501 struct timespec ts;
502 LokiEnvSensors_dma_get_packet_time(self, &ts);
503 in_timestamp = LokiTime_to_timestamp(ts) - in_period_ns;
504 }
505
506 if (!first_packet || !packet_index) {
507 // Calculer le timestamp du paquet d'entrée, en nanosecondes,
508 // en utilisant le numéro de paquet fourni par le FPGA.
509 int32_t packet_interval = packet_index - packet_index_prev;
510 if (packet_interval < 0) {
511 fprintf(stderr, "LokiEnvSensors_work: packet index rollover (%u).\n", packet_index);
512 packet_interval += packet_count_mod;
513 }
514
515 if (packet_interval > 1) {
516 fprintf(stderr, "LokiEnvSensors_work: packet index gap (%u -> %u, %u).\n", packet_index_prev, packet_index, packet_interval);
517 //goto LokiEnvSensors_work_continue;
518 }
519
520 packet_index_prev = packet_index;
521 first_packet = false;
522
523 in_timestamp += round(in_period_ns * packet_interval);
524
525 // En commençant un nouveau paquet de sortie, copier la date du paquet d'entrée.
526 if (out_ptr == (LokiEnvSensorsSample*)out_packet->buffer) {
527 out_packet->timestamp = in_timestamp;
528 }
529
530 LokiDMASample *in_ptr = in_buffer;
531 while (in_ptr != in_end && !self->stop_request) {
532 LokiDMASample in = *in_ptr ++;
533
534 LokiEnvSensorsSample s;
535 LokiEnvSensors_set_packet_item(self, &s, in);
536
537 if (self->stream_channels & (1 << s.channel)) {
538 *out_ptr ++ = s;
539
540 if (out_ptr == out_end) {
541 LokiFIFO_push(self->out);
542 out_packet = LokiFIFO_get_producer_item(self->out);
543 if (!out_packet) {
544 break;
545 }
546 if (in_ptr != in_end) {
547 out_packet->timestamp = in_timestamp + round(in_period_ns * (in_ptr - in_buffer) / in_length);
548 }
549 out_ptr = (LokiEnvSensorsSample*)out_packet->buffer;
550 out_end = out_ptr + out_length;
551 }
552 }
553 }
554 }
555
556 LokiEnvSensors_work_continue:
557 #ifdef USE_MMAP
558 munmap(in_buffer, in_length_bytes);
559 #endif
560 }
561
562 #ifndef USE_MMAP
563 free(in_buffer);
564 #endif
565
566 LokiEnvSensors_dma_stop(self);
567}

569int LokiEnvSensors_start_streaming(LokiEnvSensors *self, int32_t channels, bool raw, LokiFIFO *out) {
570 if (LokiEnvSensors_dma_is_streaming(self)) {
571 return EBUSY;
572 }
573 self->stop_request = false;
574 self->out = out;
575 self->raw = raw;
576 self->stream_channels = channels;
577 return pthread_create(&self->thread, NULL, LokiEnvSensors_work, self);
