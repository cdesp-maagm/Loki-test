57void LokiRLS_work(LokiRLS *self, float x, float y) {
58 const float out = -y * y;
59 const float in[] = {x * x + out, x * y, x, y, 1.0f};
60
61 // Copies locales de l'état du filtre.
62 float (* const P)[5] = self->P;
63 float *const state = self->state;
64 const float lambda_inv = self->lambda_inv;
65
66 // Dans les calculs suivants, on tient compte du fait que in[4] = 1
67
68 // Calcul de l'erreur courante :
69 // epsilon = out - in * state
70 float epsilon = out - state[4];
71 for (size_t i = 0; i < 4; i ++) {
72 epsilon -= in[i] * state[i];
73 }
74
75 // Calcul du produits G = P * in'
76 float G[5];
77 for (size_t i = 0; i < 5; i ++) {
78 G[i] = P[i][4];
79 for (size_t j = 0; j < 4; j ++) {
80 G[i] += P[i][j] * in[j];
81 }
82 }
83
84 // Calcul du facteur de normalisation K = lambda + in * P * in'
85 float K = self->lambda + G[4];
86 for (size_t i = 0; i < 4; i ++) {
87 K += in[i] * G[i];
88 }
89
90 // Normalisation du vecteur G et mise à jour de l'état.
91 float K_inv = 1.0f / K;
92 for (size_t i = 0; i < 5; i ++) {
93 G[i] *= K_inv;
94 state[i] += epsilon * G[i];
95 }
96
97 // Calcul de J = I - G * in
98 float J[5][5];
99 for (size_t i = 0; i < 5; i ++) {
100 J[i][i] = 1.0f - G[i] * in[i];
101 for (size_t j = i + 1; j < 5; j ++) {
102 J[i][j] = -G[i] * in[j];
103 J[j][i] = -G[j] * in[i];
104 }
105 }
106
107 // Mise à jour de la matrice P_next = J * P / lambda
108 float P_next[5][5];
109 for (size_t i = 0; i < 5; i ++) {
110 for (size_t j = 0; j < 5; j ++) {
111 P_next[i][j] = J[i][0] * P[0][j];
112 for (size_t k = 1; k < 5; k ++) {
113 P_next[i][j] += J[i][k] * P[k][j];
114 }
115 P_next[i][j] *= lambda_inv;
116 }
117 }
118
119 // Cette boucle est plus rapide que memcpy(P, P_next, sizeof(P_next)).
120 // Il est possible que le compilateur optimise le graphe flot de données
121 // sans passer par l'allocation du tableau P_next.
122 for (size_t i = 0; i < 5; i ++) {
123 for (size_t j = 0; j < 5; j ++) {
124 P[i][j] = P_next[i][j];
125 }
126 }
127}
