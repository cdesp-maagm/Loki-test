275LokiModem *LokiModem_new(void) {
276 LokiModem *self = calloc(1, sizeof(LokiModem));
277 if (!self) {
278 return NULL;
279 }
280
281 self->file = fopen("/dev/loki-modem", "r");
282 if (!self->file) {
283 fprintf(stderr, "LokiModem_new: failed to open file /dev/loki-modem (error: %d, %s)\n", errno, strerror(errno));
284 LokiModem_del(self);
285 return NULL;
286 }
287
288 self->fileno = fileno(self->file);
289
290 self->clk_frequency = LokiModem_get_reg(self, LOKI_MODEM_REG_CLK_FREQUENCY);
291 self->mod_sample_frequency_ref = LokiModem_get_reg(self, LOKI_MODEM_REG_MOD_SAMPLE_FREQUENCY_REF);
292 self->mod_cycles_per_sample_max = LokiModem_get_reg(self, LOKI_MODEM_REG_MOD_CYCLES_PER_SAMPLE_MAX);
293 self->mod_osc1_wave_frequency_min = LokiModem_get_reg(self, LOKI_MODEM_REG_MOD_OSC1_WAVE_FREQUENCY_MIN);
294 self->mod_osc2_wave_frequency_min = LokiModem_get_reg(self, LOKI_MODEM_REG_MOD_OSC2_WAVE_FREQUENCY_MIN);
295 self->mod_gain_integer_width = LokiModem_get_reg(self, LOKI_MODEM_REG_MOD_GAIN_INTEGER_WIDTH);
296 self->mod_gain_fractional_width = LokiModem_get_reg(self, LOKI_MODEM_REG_MOD_GAIN_FRACTIONAL_WIDTH);
297
298 union {
299 uint32_t u;
300 float f;
301 } mod_phase_amplitude_max, mod_dc_value_max;
302
303 mod_phase_amplitude_max.u = LokiModem_get_reg(self, LOKI_MODEM_REG_MOD_PHASE_AMPLITUDE_MAX);
304 self->mod_phase_amplitude_max = mod_phase_amplitude_max.f;
305 self->mod_phase_integer_width = LokiModem_get_reg(self, LOKI_MODEM_REG_MOD_PHASE_INTEGER_WIDTH);
306 self->mod_phase_fractional_width = LokiModem_get_reg(self, LOKI_MODEM_REG_MOD_PHASE_FRACTIONAL_WIDTH);
307 mod_dc_value_max.u = LokiModem_get_reg(self, LOKI_MODEM_REG_MOD_DC_VALUE_MAX);
308 self->mod_dc_value_max = mod_dc_value_max.f;
309 self->mod_dc_value_width = LokiModem_get_reg(self, LOKI_MODEM_REG_MOD_DC_VALUE_WIDTH);
310 self->mod_pga_gain_width = LokiModem_get_reg(self, LOKI_MODEM_REG_MOD_PGA_GAIN_WIDTH);
311 self->demod_mixerless = !!(LokiModem_get_reg(self, LOKI_MODEM_REG_CONTROL) & LOKI_MODEM_REG_CONTROL_MIXERLESS_SHIFTED_MASK);
312 self->demod_channels = LokiModem_get_reg(self, LOKI_MODEM_REG_DEMOD_CHANNELS);
313 self->demod_channel_output_width = LokiModem_get_reg(self, LOKI_MODEM_REG_DEMOD_CHANNEL_OUTPUT_WIDTH);
314 self->demod_power_integer_width = LokiModem_get_reg(self, LOKI_MODEM_REG_DEMOD_POWER_INTEGER_WIDTH);
315 self->demod_power_fractional_width = LokiModem_get_reg(self, LOKI_MODEM_REG_DEMOD_POWER_FRACTIONAL_WIDTH);
316 self->demod_sample_frequency_ref = LokiModem_get_reg(self, LOKI_MODEM_REG_DEMOD_SAMPLE_FREQUENCY_REF);
317 self->demod_cycles_per_sample_max = LokiModem_get_reg(self, LOKI_MODEM_REG_DEMOD_CYCLES_PER_SAMPLE_MAX);
318 self->demod_osc_wave_frequency_min = LokiModem_get_reg(self, LOKI_MODEM_REG_DEMOD_OSC_WAVE_FREQUENCY_MIN);
319 self->demod_fir1_decimation = LokiModem_get_reg(self, LOKI_MODEM_REG_DEMOD_FIR1_DECIMATION);
320 self->demod_fir1_length = LokiModem_get_reg(self, LOKI_MODEM_REG_DEMOD_FIR1_LENGTH);
321 self->demod_fir2_decimation = LokiModem_get_reg(self, LOKI_MODEM_REG_DEMOD_FIR2_DECIMATION);
322 self->demod_fir2_length = LokiModem_get_reg(self, LOKI_MODEM_REG_DEMOD_FIR2_LENGTH);
323 self->demod_fir2_bandwidth = LokiModem_get_reg(self, LOKI_MODEM_REG_DEMOD_FIR2_BANDWIDTH);
324 self->demod_xy_integer_width = LokiModem_get_reg(self, LOKI_MODEM_REG_DEMOD_XY_INTEGER_WIDTH);
325 self->demod_xy_fractional_width = LokiModem_get_reg(self, LOKI_MODEM_REG_DEMOD_XY_FRACTIONAL_WIDTH);
326
327 self->demod_channel_mask = (1 << self->demod_channel_output_width) - 1;
328
329 self->mod_cycles_per_sample_ref = (float)self->clk_frequency / self->mod_sample_frequency_ref;
330 self->mod_gain_res = self->mod_phase_amplitude_max / (1 << (self->mod_gain_integer_width + self->mod_gain_fractional_width - 1) - 1);
331 self->mod_offset_res = self->mod_phase_amplitude_max / (1 << (self->mod_phase_integer_width + self->mod_phase_fractional_width - 1) - 1);
332 self->mod_dc_value_res = self->mod_dc_value_max / ((1 << self->mod_dc_value_width) - 1);
333 self->mod_pga_gain_max = (1 << self->mod_pga_gain_width) - 1;
334 self->demod_power_max = LokiModem_scale_power(self, ((1 << (self->demod_power_integer_width + self->demod_power_fractional_width - 1))) - 1);
335 self->demod_cycles_per_sample_ref = (float)self->clk_frequency / self->demod_sample_frequency_ref;
336 self->demod_xy_max = LokiModem_scale_xy(self, ((1 << (self->demod_xy_integer_width + self->demod_xy_fractional_width - 1))) - 1);
337 self->demod_xy_width = self->demod_xy_integer_width + self->demod_xy_fractional_width;
338 self->demod_xy_mask = (1 << self->demod_xy_width) - 1;
339 self->demod_xy_sign_mask = 1 << (self->demod_xy_width - 1);
340
341 if (!LokiModem_is_streaming(self)) {
342 // Le nombre d'octets d'un transfert DMA doit être multiple de
343 // * la taille d'une page mémoire
344 // * la taille d'un échantillon
345 uint32_t page_size = getpagesize();
346 uint32_t a = page_size;
347 uint32_t b = sizeof(LokiDMASample);
348 while (a != b) {
349 if (a > b) {
350 a -= b;
351 }
352 if (b > a) {
353 b -= a;
354 }
355 }
356
357 uint32_t packet_len_min = page_size / a;
358 uint32_t packet_len_max = LokiModem_get_demod_packet_length_max(self);
359 packet_len_max -= packet_len_max % packet_len_min;
360
361 uint32_t packet_len = self->demod_channels * packet_len_min;
362 if (packet_len > packet_len_max) {
363 packet_len = packet_len_max;
364 }
365
366 LokiModem_set_reg(self, LOKI_MODEM_REG_DEMOD_PACKET_LENGTH, packet_len);
367 }
368
369 return self;
370}

836static void LokiModem_set_packet_item(const LokiModem *self, LokiModemSample *item, LokiDMASample raw_item) {
837 // Lire la valeur de l'échantillon, alignée à droite en complément à deux.
838 // Effectuer l'extension du bit de signe.
839 int32_t y_raw = raw_item & self->demod_xy_mask;
840 raw_item >>= self->demod_xy_width;
841
842 if (y_raw & self->demod_xy_sign_mask) {
843 y_raw |= ~self->demod_xy_mask;
844 }
845 else {
846 y_raw &= self->demod_xy_mask;
847 }
848
849 int32_t x_raw = raw_item & self->demod_xy_mask;
850 raw_item >>= self->demod_xy_width;
851
852 if (x_raw & self->demod_xy_sign_mask) {
853 x_raw |= ~self->demod_xy_mask;
854 }
855 else {
856 x_raw &= self->demod_xy_mask;
857 }
858
859 // Convertir les coordonnées en flottants si nécessaire.
860 if (self->raw) {
861 item->x_raw = x_raw;
862 item->y_raw = y_raw;
863 }
864 else {
865 item->x = LokiModem_scale_xy(self, x_raw);
866 item->y = LokiModem_scale_xy(self, y_raw);
867 }
868
869 // Lire le numéro de la voie.
870 item->channel = raw_item & self->demod_channel_mask;
871}

943static void *LokiModem_work(void *ptr) {
944 LokiModem *self = ptr;
945
946 pthread_setname_np(self->thread, "LokiModem");
947
948 // Obtenir un pointeur vers le buffer courant des échantillons mise en forme.
949 const size_t out_length = LokiPacket_get_length(self->out, sizeof(LokiModemSample));
950 LokiPacket *out_packet = LokiFIFO_get_producer_item(self->out);
951 if (!out_packet) {
952 return NULL;
953 }
954 LokiModemSample *out_ptr = (LokiModemSample*)out_packet->buffer;
955 LokiModemSample *out_end = out_ptr + out_length;
956
957 // Avec la fonction fread, allouer un buffer de données.
958 // Avec l'accès par mmap, le buffer est fourni par le pilote.
959 const size_t in_length = LokiModem_get_demod_packet_length(self);
960 #ifndef USE_MMAP
961 LokiDMASample *in_buffer = calloc(in_length, sizeof(LokiDMASample));
962 LokiDMASample *in_end = in_buffer + in_length;
963 #else
964 size_t in_length_bytes = in_length * sizeof(LokiDMASample);
965 #endif
966
967 // Calculer la durée d'un paquet d'entrée, en nanosecondes.
968 double in_period_ns = 1.0e9 * in_length / self->demod_channels / LokiModem_get_demod_xy_sampling_frequency(self);
969
970 // Calculer le modulo du compteur de paquets du FPGA.
971 const uint32_t packet_count_mod = 1 << (sizeof(LokiDMASample) * 8 - self->demod_channel_output_width - self->demod_xy_width * 2);
972
973 bool first_packet = true;
974 uint32_t packet_index_prev = 0;
975 LokiTimestamp in_timestamp;
976
977 // Démarrer l'acquisition.
978 LokiModem_dma_start(self);
979
980 while (!self->stop_request) {
981 // Lire un buffer d'échantillons en provenance du DMA.
982 #ifndef USE_MMAP
983 if (fread(in_buffer, sizeof(LokiDMASample), in_length, self->file) != in_length) {
984 fprintf(stderr, "LokiModem_work: read failed (error: %d, %s)\n", errno, strerror(errno));
985 break;
986 }
987 #else
988 // Accès au buffer DMA en mode zero-copy.
989 LokiDMASample *in_buffer = mmap(NULL, in_length_bytes, PROT_READ, MAP_PRIVATE, self->fileno, 0);
990 if (in_buffer == MAP_FAILED) {
991 fprintf(stderr, "LokiModem_work: mmap failed (error: %d, %s)\n", errno, strerror(errno));
992 break;
993 }
994 LokiDMASample *in_end = in_buffer + in_length;
995 #endif
996
997 // Attendre le paquet numéro 0 pour commencer.
998 uint32_t packet_index = LokiModem_get_packet_index(self, *in_buffer);
999
1000 // Prendre pour référence temporelle la date du premier paquet.
1001 // Les dates des autres paquets seront calculées en fonction du nombre d'échantillons.
1002 if (first_packet && !packet_index) {
1003 struct timespec ts;
1004 LokiModem_dma_get_packet_time(self, &ts);
1005 in_timestamp = LokiTime_to_timestamp(ts) - in_period_ns;
1006 }
1007
1008 if (!first_packet || !packet_index) {
1009 // Calculer le timestamp du paquet d'entrée, en nanosecondes,
1010 // en utilisant le numéro de paquet fourni par le FPGA.
1011 int32_t packet_interval = packet_index - packet_index_prev;
1012 if (packet_interval < 0) {
1013 fprintf(stderr, "LokiModem_work: packet index rollover (%u -> %u).\n", packet_index_prev, packet_index);
1014 packet_interval += packet_count_mod;
1015 }
1016
1017 if (packet_interval < 0) {
1018 fprintf(stderr, "LokiModem_work: packet index overlap (%u -> %u, %u).\n", packet_index_prev, packet_index, -packet_interval);
1019 }
1020 else if (packet_interval > 1) {
1021 fprintf(stderr, "LokiModem_work: packet index gap (%u -> %u, %u).\n", packet_index_prev, packet_index, packet_interval);
1022 //goto LokiModem_work_continue;
1023 }
1024
1025 packet_index_prev = packet_index;
1026 first_packet = false;
1027
1028 in_timestamp += round(in_period_ns * packet_interval);
1029
1030 // En commençant un nouveau paquet de sortie, copier la date du paquet d'entrée.
1031 if (out_ptr == (LokiModemSample*)out_packet->buffer) {
1032 out_packet->timestamp = in_timestamp;
1033 }
1034
1035 LokiDMASample *in_ptr = in_buffer;
1036 while (in_ptr != in_end && !self->stop_request) {
1037 LokiDMASample in = *in_ptr ++;
1038
1039 LokiModemSample s;
1040 if (self->simulate) {
1041 LokiModem_set_simulated_item(self, &s);
1042 }
1043 else {
1044 LokiModem_set_packet_item(self, &s, in);
1045 }
1046
1047 // Produire l'échantillon en sortie si la voie correspondante
1048 // est activée dans le masque des voies à traiter.
1049 if (self->stream_channels & (1 << s.channel)) {
1050 *out_ptr ++ = s;
1051
1052 if (out_ptr == out_end) {
1053 LokiFIFO_push(self->out);
1054 out_packet = LokiFIFO_get_producer_item(self->out);
1055 if (!out_packet) {
1056 break;
1057 }
1058 if (in_ptr != in_end) {
1059 out_packet->timestamp = in_timestamp + round(in_period_ns * (in_ptr - in_buffer) / in_length);
1060 }
1061 out_ptr = (LokiModemSample*)out_packet->buffer;
1062 out_end = out_ptr + out_length;
1063 }
1064 }
1065 }
1066 }
1067
1068 LokiModem_work_continue:
1069 #ifdef USE_MMAP
1070 munmap(in_buffer, in_length_bytes);
1071 #endif
1072 }
1073
1074 #ifndef USE_MMAP
1075 free(in_buffer);
1076 #endif
1077
1078 LokiModem_dma_stop(self);
1079}
