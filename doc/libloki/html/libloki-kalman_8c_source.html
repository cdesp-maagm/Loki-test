60void LokiKalman_work(LokiKalman *self, float x, float y) {
61 const float out = -y * y;
62 const float in[] = {x * x + out, x * y, x, y, 1.0f};
63
64 // Copies locales de l'état du filtre.
65 float (* const P)[5] = self->P;
66 float *const state = self->state;
67 float rw = self->rw;
68
69 // Dans les calculs suivants, on tient compte du fait que in[4] = 1
70
71 // Calcul de l'erreur courante :
72 // epsilon = out - in * state
73 float epsilon = out - state[4];
74 for (size_t i = 0; i < 4; i ++) {
75 epsilon -= in[i] * state[i];
76 }
77
78 // Calcul du produits L = P * in'
79 float L[5];
80 for (size_t i = 0; i < 5; i ++) {
81 L[i] = P[i][4];
82 for (size_t j = 0; j < 4; j ++) {
83 L[i] += P[i][j] * in[j];
84 }
85 }
86
87 // Calcul du dénominateur K = Rv + in * P * in'
88 float K = self->rv + L[4];
89 for (size_t i = 0; i < 4; i ++) {
90 K += in[i] * L[i];
91 }
92
93 // Normalisation du vecteur L et mise à jour de l'état.
94 float K_inv = 1.0f / K;
95 for (size_t i = 0; i < 5; i ++) {
96 L[i] *= K_inv;
97 state[i] += epsilon * L[i];
98 }
99
100 // Calcul de J = I - L * in
101 float J[5][5];
102 for (size_t i = 0; i < 5; i ++) {
103 J[i][i] = 1.0f - L[i] * in[i];
104 for (size_t j = i + 1; j < 5; j ++) {
105 J[i][j] = -L[i] * in[j];
106 J[j][i] = -L[j] * in[i];
107 }
108 }
109
110 // Mise à jour de la matrice P_next = J * P + Rw
111 float P_next[5][5];
112 for (size_t i = 0; i < 5; i ++) {
113 for (size_t j = 0; j < 5; j ++) {
114 P_next[i][j] = J[i][0] * P[0][j];
115 for (size_t k = 1; k < 5; k ++) {
116 P_next[i][j] += J[i][k] * P[k][j];
117 }
118 }
119 P_next[i][i] += rw;
120 }
121
122 // Cette boucle est plus rapide que memcpy(P, P_next, sizeof(P_next)).
123 // Il est possible que le compilateur optimise le graphe flot de données
124 // sans passer par l'allocation du tableau P_next.
125 for (size_t i = 0; i < 5; i ++) {
126 for (size_t j = 0; j < 5; j ++) {
127 P[i][j] = P_next[i][j];
128 }
129 }
130}
