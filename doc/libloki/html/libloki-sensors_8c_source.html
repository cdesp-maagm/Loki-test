173 LokiSensors *LokiSensors_new(void) {
174  LokiSensors *self = calloc(1, sizeof(LokiSensors));
175  if (!self) {
176  return NULL;
177  }
178 
179  self->file = fopen("/dev/loki-adc", "r");
180  if (!self->file) {
181  LokiSensors_del(self);
182  return NULL;
183  }
184 
185  self->fileno = fileno(self->file);
186 
187  self->clk_frequency = LokiSensors_get_reg(self, LOKI_ADC_REG_CLK_FREQUENCY);
188  self->channels = LokiSensors_get_reg(self, LOKI_ADC_REG_CHANNELS);
189  self->channel_output_width = LokiSensors_get_reg(self, LOKI_ADC_REG_CHANNEL_OUTPUT_WIDTH);
190  self->data_width = LokiSensors_get_reg(self, LOKI_ADC_REG_DATA_WIDTH);
191  self->cycles_per_sample_max = LokiSensors_get_reg(self, LOKI_ADC_REG_CYCLES_PER_SAMPLE_MAX);
192  self->packet_length_max = LokiSensors_get_reg(self, LOKI_ADC_REG_PACKET_LENGTH_MAX);
193  self->channel_mask = (1 << self->channel_output_width) - 1;
194  self->data_mask = (1 << self->data_width) - 1;
195  self->data_sign_mask = 1 << (self->data_width - 1);
196 
197  // Allouer un tableau pour les facteurs de mise à l'échelle,
198  // et les initialiser à 1.
199  self->scales = calloc(self->channels, sizeof(float));
200  if (!self->scales) {
201  LokiSensors_del(self);
202  return NULL;
203  }
204 
205  for (size_t i = 0; i < self->channels; i ++) {
206  self->scales[i] = 1.0f;
207  }
208 
209  // Le nombre d'octets d'un transfert DMA doit être multiple de
210  // * la taille d'une page mémoire
211  // * le nombre de voies * la taille d'un échantillon
212  uint32_t page_size = getpagesize();
213  uint32_t a = page_size;
214  uint32_t b = self->channels * sizeof(LokiDMASample);
215  while (a != b) {
216  if (a > b) {
217  a -= b;
218  }
219  if (b > a) {
220  b -= a;
221  }
222  }
223 
224  self->packet_length_min = page_size / a;
225  self->packet_length_max -= self->packet_length_max % self->packet_length_min;
226 
227  LokiSensors_set_packet_length(self, LokiSensors_get_packet_length(self));
228 
229  return self;
230 }

321 static void LokiSensors_set_packet_item(const LokiSensors *self, LokiSensorsSample *item, LokiDMASample raw_item) {
322  // Lire la valeur de l'échantillon, alignée à droite en complément à deux.
323  // Effectuer l'extension du bit de signe.
324  int32_t data_raw = raw_item & self->data_mask;
325  raw_item >>= self->data_width;
326 
327  if (data_raw & self->data_sign_mask) {
328  data_raw |= ~self->data_mask;
329  }
330  else {
331  data_raw &= self->data_mask;
332  }
333 
334  // Convertir les coordonnées en flottants si nécessaire.
335  if (self->raw) {
336  item->data_raw = data_raw;
337  }
338  else {
339  item->data = self->scales[item->channel] * data_raw;
340  }
341 
342  // Lire le numéro de la voie.
343  item->channel = raw_item & self->channel_mask;
344 }

368 static void *LokiSensors_work(void *ptr) {
369  LokiSensors *self = ptr;
370 
371  pthread_setname_np(self->thread, "LokiSensors");
372 
373  // Obtenir un pointeur vers le buffer courant des échantillons mise en forme.
374  const size_t out_length = LokiPacket_get_length(self->out, sizeof(LokiSensorsSample));
375  LokiPacket *out_packet = LokiFIFO_get_producer_item(self->out);
376  if (!out_packet) {
377  return NULL;
378  }
379  LokiSensorsSample *out_ptr = (LokiSensorsSample*)out_packet->buffer;
380  LokiSensorsSample *out_end = out_ptr + out_length;
381 
382  // Avec la fonction fread, allouer un buffer de données.
383  // Avec l'accès par mmap, le buffer est fourni par le pilote.
384  const size_t in_length = LokiSensors_get_transfer_length(self);
385  #ifndef USE_MMAP
386  LokiDMASample *in_buffer = calloc(in_length, sizeof(LokiDMASample));
387  LokiDMASample *in_end = in_buffer + in_length;
388  #else
389  size_t in_length_bytes = in_length * sizeof(LokiDMASample);
390  #endif
391 
392  // Calculer la durée d'un paquet d'entrée, en nanosecondes.
393  double in_period_ns = 1.0e9 * in_length / self->channels / LokiSensors_get_sampling_frequency(self);
394 
395  // Calculer le modulo du compteur de paquets du FPGA.
396  uint32_t packet_count_mod = 1 << (sizeof(LokiDMASample) * 8 - self->channel_output_width - self->data_width);
397 
398  bool first_packet = true;
399  uint32_t packet_index_prev = 0;
400  LokiTimestamp in_timestamp;
401 
402  // Démarrer l'acquisition.
403  LokiSensors_dma_start(self);
404 
405  while (!self->stop_request) {
406  // Lire un buffer d'échantillons en provenance du DMA.
407  #ifndef USE_MMAP
408  if (fread(in_buffer, sizeof(LokiDMASample), in_length, self->file) != in_length) {
409  fprintf(stderr, "LokiSensors_work: read failed (error: %d, %s)\n", errno, strerror(errno));
410  break;
411  }
412  #else
413  // Accès au buffer DMA en mode zero-copy.
414  // Contrairement à l'acquisition des voies optiques, l'appel à mmap
415  // peut bloquer le processus pendant une longue durée.
416  // On utilise le timeout pour le débloquer jusqu'au succès.
417  LokiDMASample *in_buffer;
418  do {
419  in_buffer = mmap(NULL, in_length_bytes, PROT_READ, MAP_PRIVATE, self->fileno, 0);
420  } while (in_buffer == MAP_FAILED && errno == ETIMEDOUT);
421 
422  if (in_buffer == MAP_FAILED) {
423  fprintf(stderr, "LokiSensors_work: mmap failed (error: %d, %s)\n", errno, strerror(errno));
424  break;
425  }
426 
427  LokiDMASample *in_end = in_buffer + in_length;
428  #endif
429 
430  // Convertir la date du paquet reçu en timestamp.
431  struct timespec ts;
432  LokiSensors_dma_get_packet_time(self, &ts);
433  LokiTimestamp timestamp = LokiTime_to_timestamp(ts);
434 
435  // Attendre le paquet numéro 0 pour commencer.
436  uint32_t packet_index = LokiSensors_get_packet_index(self, *in_buffer);
437 
438  // Prendre pour référence temporelle la date du premier paquet.
439  // Les dates des autres paquets seront calculées en fonction du nombre d'échantillons.
440  if (first_packet /* && !packet_index */) { // TODO ajouter une réinitialisation du numéro de paquet dans LTC2325_AXI
441  in_timestamp = timestamp;
442  }
443 
444  if (true /* !first_packet || !packet_index */) { // TODO ajouter une réinitialisation du numéro de paquet dans LTC2325_AXI
445  // Calculer le timestamp du paquet d'entrée, en nanosecondes,
446  // en utilisant le numéro de paquet fourni par le FPGA.
447  int32_t packet_interval = packet_index - packet_index_prev;
448  packet_index_prev = packet_index;
449  first_packet = false;
450 
451  if (packet_interval < 0) {
452  fprintf(stderr, "LokiSensors_work: packet index rollover (%u).\n", packet_index);
453  packet_interval += packet_count_mod;
454  // Ignorer le paquet si l'inversion de chronologie est liée à une perte de paquet.
455  if (packet_interval > 1) {
456  fprintf(stderr, "LokiSensors_work: discarding packet.\n");
457  goto LokiSensors_work_continue;
458  }
459  }
460 
461  in_timestamp += round(in_period_ns * packet_interval);
462 
463  // En commençant un nouveau paquet de sortie, copier la date du paquet d'entrée.
464  if (out_ptr == (LokiSensorsSample*)out_packet->buffer) {
465  out_packet->timestamp = in_timestamp;
466  }
467 
468  LokiDMASample *in_ptr = in_buffer;
469  while (in_ptr != in_end && !self->stop_request) {
470  LokiDMASample in = *in_ptr ++;
471 
472  LokiSensorsSample s;
473  LokiSensors_set_packet_item(self, &s, in);
474 
475  if (self->stream_channels & (1 << s.channel)) {
476  *out_ptr ++ = s;
477 
478  if (out_ptr == out_end) {
479  LokiFIFO_push(self->out);
480  out_packet = LokiFIFO_get_producer_item(self->out);
481  if (!out_packet) {
482  break;
483  }
484  if (in_ptr != in_end) {
485  out_packet->timestamp = in_timestamp + round(in_period_ns * (in_ptr - in_buffer) / in_length);
486  }
487  out_ptr = (LokiSensorsSample*)out_packet->buffer;
488  out_end = out_ptr + out_length;
489  }
490  }
491  }
492  }
493 
494  LokiSensors_work_continue:
495  #ifdef USE_MMAP
496  munmap(in_buffer, in_length_bytes);
497  #endif
498  }
499 
500  #ifndef USE_MMAP
501  free(in_buffer);
502  #endif
503 
504  LokiSensors_dma_stop(self);
505 }
