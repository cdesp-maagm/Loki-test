60LokiFIFO *LokiFIFO_new(size_t item_size, size_t length, size_t consumers) {
61 size_t record_size = sizeof(LokiFIFORecord) + item_size;
62 LokiFIFO *self = calloc(1, sizeof(LokiFIFO) + length * record_size + consumers * sizeof(LokiFIFORecord*));
63 if (!self) {
64 return NULL;
65 }
66
67 LokiFIFORecord *first_record = (LokiFIFORecord*)(self->consumer_records + consumers);
68
69 self->abort = false;
70 self->item_size = item_size;
71 self->all_read = (1 << consumers) - 1;
72 self->producer_record = first_record;
73 for (size_t c = 0; c < consumers; c ++) {
74 self->consumer_records[c] = first_record;
75 }
76
77 // Initialiser les records comme une liste chaînée circulaire.
78 // Marquer tous les éléments comme lus par tous les consommateurs.
79 LokiFIFORecord *rec = first_record;
80 for (size_t i = 0; i < length; i ++) {
81 rec->status = self->all_read;
82 rec->next = i == length - 1 ?
83 first_record :
84 (LokiFIFORecord*)((char*)rec + record_size);
85 rec = rec->next;
86 }
87
88 pthread_mutex_init(&self->mutex, NULL);
89 pthread_cond_init(&self->pushed, NULL);
90 pthread_cond_init(&self->pulled, NULL);
91
92 return self;
93}

123void LokiFIFO_push(LokiFIFO *self) {
124 if (self->abort) {

126 }

128 pthread_mutex_lock(&self->mutex);

130 // Marquer l'élément courant comme non lu.
