295LokiSystem *LokiSystem_new(LokiSystemConfig *config, LokiModem *modem) {
296 #ifdef DEBUG_MALLOC
297 fprintf(stderr, "LokiSystem_new: Starting memory allocation monitoring\n");
298 LokiHooks_init();
299 #endif
300
301 // Vérifier les paramètres de configuration.
302 if (!config->estimator_impl) {
303 return NULL;
304 }
305
306 LokiSystem *self = calloc(1, sizeof(LokiSystem));
307 if (!self) {
308 return NULL;
309 }
310
311 self->config = config;
312
313 // Initialiser et configurer le modem
314 self->has_own_modem = modem == NULL;
315 self->modem = self->has_own_modem ? LokiModem_new() : modem;
316 if (!self->modem) {
317 LokiSystem_del(self);
318 return NULL;
319 }
320
321 self->channels = LokiModem_get_demod_channels(self->modem);
322
323 // Initialiser la chaîne de traitement du signal logiciel.
324 self->disp = calloc(self->channels, sizeof(LokiDisplacement*));
325 if (!self->disp) {
326 LokiSystem_del(self);
327 return NULL;
328 }
329
330 LokiEllipseEstimator *estimator = LokiEllipseEstimator_new_with_implementation(config->estimator_impl, config->estimator_initial_radius, config->estimator_params);
331 LokiFIRFilter *filter = LokiFIRFilter_new_kaiser_bessel(LokiModem_get_demod_xy_sampling_frequency(self->modem), 0, config->filter_bandwidth, config->filter_attenuation, config->filter_length, config->filter_decimation);
332
333 float scale = LokiSystemConfig_get_scale(config);
334 float sampling_frequency = LokiModem_get_demod_xy_sampling_frequency(self->modem);
335
336 for (size_t c = 0; c < self->channels; c ++) {
337 self->disp[c] = c == 0 ?
338 LokiDisplacement_new(estimator, filter, config->estimator_throttle, scale, config->time_constants, sampling_frequency) :
339 LokiDisplacement_clone(self->disp[0]);
340 if (!self->disp[c]) {
341 LokiSystem_del(self);
342 return NULL;
343 }
344 }
345
346 pthread_mutex_init(&self->mutex, NULL);
347
348 return self;
349}

380static void *LokiSystem_work(void *ptr) {
381 LokiSystem *self = ptr;
382
383 pthread_setname_np(self->thread, "LokiSystem");
384
385 // La taille des buffers de données dans les FIFOs d'entrée et de sortie, en échantillons.
386 const size_t in_length = LokiPacket_get_length(self->in, sizeof(LokiModemSample));
387 const size_t out_length = LokiPacket_get_length(self->out, sizeof(LokiSystemSample));
388
389 // Références vers le premier paquet et le premier échantillon de la FIFO des déplacements,
390 // qui ne progresse pas à la même vitesse que la FIFO des échantillons du modem.
391 LokiPacket *out_packet = LokiFIFO_get_producer_item(self->out);
392 if (!out_packet) {
393 goto LokiSystem_work_end;
394 }
395 LokiSystemSample *out_ptr = (LokiSystemSample*)out_packet->buffer;
396 LokiSystemSample *out_end = out_ptr + out_length;
397
398 // Calculer la durée d'un paquet d'entrée, en nanosecondes.
399 const size_t channels = self->channels;
400 double in_period_ns = 1.0e9 * in_length / channels / LokiModem_get_demod_xy_sampling_frequency(self->modem);
401
402 // Préparer un tableau avec l'indice de l'échantillon courant pour chaque voie.
403 uint64_t skip = self->config->skip * LokiSystem_get_sampling_frequency(self);
404 uint64_t *sample_index = calloc(channels, sizeof(uint64_t));
405 if (!sample_index) {
406 goto LokiSystem_work_end;
407 }
408
409 // Préparer des tableaux pour mémoriser les points de l'ellipse normalisée.
410 const size_t decimation = self->config->filter_decimation;
411 float *xy_norm = calloc(channels * decimation * 2, sizeof(float));
412 if (!xy_norm) {
413 goto LokiSystem_work_end;
414 }
415 float **xy_start = calloc(channels * 4, sizeof(float*));
416 if (!xy_start) {
417 goto LokiSystem_work_end;
418 }
419
420 float **xy_end = xy_start + channels;
421 float **xy_read = xy_start + channels * 2;
422 float **xy_write = xy_start + channels * 3;
423 for (size_t c = 0; c < channels; c ++) {
424 xy_start[c] = xy_read[c] = xy_write[c] = xy_norm + (c * decimation * 2);
425 xy_end[c] = xy_start[c] + (decimation * 2);
426 }
427
428 while (!self->stop_request) {
429 LokiPacket *in_packet = LokiFIFO_get_consumer_item(self->in, self->cid);
430 if (!in_packet) {
431 break;
432 }
433 LokiModemSample *in_ptr = (LokiModemSample*)in_packet->buffer;
434 const LokiModemSample *const in_end = in_ptr + in_length;
435
436 // En commençant un nouveau paquet de sortie, copier la date du paquet d'entrée.
437 if (out_ptr == (LokiSystemSample*)out_packet->buffer) {
438 out_packet->timestamp = in_packet->timestamp;
439 }
440
441 // Calculer les déplacements, remplir les FIFO de sortie.
442 while (in_ptr != in_end && !self->stop_request) {
443 LokiModemSample in = *in_ptr ++;
444
445 if (in.channel >= channels) {
446 continue;
447 }
448
449 pthread_mutex_lock(&self->mutex);
450 float d;
451 bool has_disp = LokiDisplacement_work(self->disp[in.channel], in.x, in.y, xy_write[in.channel], xy_write[in.channel] + 1, &d);
452 pthread_mutex_unlock(&self->mutex);
453
454 xy_write[in.channel] += 2;
455 if (xy_write[in.channel] == xy_end[in.channel]) {
456 xy_write[in.channel] = xy_start[in.channel];
457 }
458
459 if (has_disp) {
460 sample_index[in.channel] ++;
461
462 // Ne pas remplir le buffer de sortie tant que le nombre
463 // d'échantillons à omettre n'est pas atteint.
464 if (sample_index[in.channel] < skip) {
465 continue;
466 }
467
468 // Remettre à zéro le déplacement courant avant d'enregistrer.
469 if (sample_index[in.channel] == skip) {
470 LokiDisplacement_reset(self->disp[in.channel]);
471 d = 0;
472 }
473
474 *out_ptr ++ = (LokiSystemSample) {
475 .channel = in.channel,
476 .value = d,
477 .x_norm = xy_read[in.channel][0],
478 .y_norm = xy_read[in.channel][1]
479 };
480
481 xy_read[in.channel] += 2;
482 if (xy_read[in.channel] == xy_end[in.channel]) {
483 xy_read[in.channel] = xy_start[in.channel];
484 }
485
486 if (out_ptr == out_end) {
487 LokiFIFO_push(self->out);
488 out_packet = LokiFIFO_get_producer_item(self->out);
489 if (!out_packet) {
490 goto LokiSystem_work_end;
491 }
492 if (in_ptr != in_end) {
493 out_packet->timestamp = in_packet->timestamp + round(in_period_ns * (in_ptr - (LokiModemSample*)in_packet->buffer) / in_length);
494 }
495 out_ptr = (LokiSystemSample*)out_packet->buffer;
496 out_end = out_ptr + out_length;
497 }
498 }
499 }
500
501 LokiFIFO_pull(self->in, self->cid);
502 }
503
504LokiSystem_work_end:
505 LokiModem_stop_streaming(self->modem);
506
507 if (self->stop_request) {
508 fprintf(stderr, "LokiSystem_work: complete\n");
509 }
510 else {
511 fprintf(stderr, "LokiSystem_work: aborted\n");
512 }
513
514 if (sample_index) {
515 free(sample_index);
516 }
517 if (xy_norm) {
518 free(xy_norm);
519 }
520 if (xy_start) {
521 free(xy_start);
522 }
523
524 self->is_streaming = false;
525}

527int LokiSystem_start_streaming(LokiSystem *self, int32_t channels, LokiFIFO *modem_out, LokiFIFO *disp_out, size_t cid) {
528 if (self->is_streaming) {
529 return EBUSY;
530 }
531
532 self->in = modem_out;
533 self->cid = cid;
534 self->out = disp_out;
535 self->stop_request = false;
536 self->is_streaming = true;
537
538 if (disp_out) {
539 int err = pthread_create(&self->thread, NULL, LokiSystem_work, self);
540 if (err) {
541 return err;
542 }
543 }
544
545 LokiModem_restart(self->modem);
546
547 return LokiModem_start_streaming(self->modem, channels, self->config->simulate, false, modem_out);
548}

550void LokiSystem_stop_streaming(LokiSystem *self) {
551 if (self->stop_request) {
552 return;
553 }
554 self->stop_request = true;
555 if (self->out) {
556 LokiFIFO_abort(self->in);
557 LokiFIFO_abort(self->out);
558 pthread_join(self->thread, NULL);
559 }
560 // Si le thread n'a pas été démarré, il faut
561 // désactiver cet indicateur ici.
562 self->is_streaming = false;
563}
