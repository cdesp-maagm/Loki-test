56void LokiNLMS_work(LokiNLMS *self, float x, float y) {
57 float out = -y * y;
58 float in[] = {x * x + out, x * y, x, y, 1};
59
60 // Calcul de l'erreur courante :
61 // epsilon = out - in * state
62 float epsilon = out;
63 for (size_t i = 0; i < LEN(in); i ++) {
64 epsilon -= in[i] * self->state[i];
65 }
66
67 // Mise à jour des facteurs de normalisation
68 // en utilisant un filtre IIR du premier ordre.
69 // Mise à jour des coefficients par descente de gradient.
70 float alpha = self->alpha;
71 float umalpha = 1.0f - alpha;
72 float m = self->mu * epsilon;
73 for (size_t i = 0; i < LEN(in); i ++) {
74 self->weights[i] = umalpha * self->weights[i] + alpha * in[i] * in[i];
75 self->state[i] += m * in[i] / (self->weights[i] + NLMS_NU);
76 }
77}
