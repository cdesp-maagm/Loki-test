77static void *LokiSocketServer_receive(void *ptr) {
78 LokiSocketServer *self = ptr;
79
80 pthread_setname_np(self->receiver_thread, "LokiSocketRecv");
81
82 self->data_size = LokiPacket_get_length(self->in, 1);
83
84 while (!self->stop_request) {
85 // Lire un paquet de données.
86 LokiPacket *in_packet = LokiFIFO_get_consumer_item(self->in, self->cid);
87 if (!in_packet) {
88 break;
89 }
90
91 // Si une connexion est ouverte, indiquer la disponibilité
92 // des données au thread serveur.
93 pthread_mutex_lock(&self->mutex);
94 self->data = in_packet->buffer;
95 self->has_data = true;
96 pthread_cond_signal(&self->has_data_cond);
97 pthread_mutex_unlock(&self->mutex);
98
99 LokiFIFO_pull(self->in, self->cid);
100 }
101
102 self->is_streaming = false;

114static void *LokiSocketServer_serve(void *ptr) {
115 LokiSocketServer *self = ptr;
116
117 pthread_setname_np(self->server_thread, "LokiSocketServe");
118
119 // Créer une socket.
120 int sock = socket(AF_UNIX, SOCK_STREAM | SOCK_NONBLOCK, 0);
121 bool has_file = false;
122 bool listening = false;
123 int conn[LIBLOKI_SOCKET_SERVER_CONNECTIONS];
124 bool status[LIBLOKI_SOCKET_SERVER_CONNECTIONS] = {false};
125 size_t accept_index = 0;
126
127 if (sock < 0) {
128 fprintf(stderr, "LokiSocketServer_serve (%s): socket creation failed (%s).\n", self->filename, strerror(errno));
129 }
130 else {
131 // Attacher cette socket à un nom de fichier.
132 struct sockaddr_un addr;
133 memset(&addr, 0, sizeof(addr));
134 addr.sun_family = AF_UNIX;
135 strncpy(addr.sun_path, self->filename, sizeof(addr.sun_path) - 1);
136 has_file = !bind(sock, (struct sockaddr*)&addr, sizeof(addr));
137 if (!has_file) {
138 fprintf(stderr, "LokiSocketServer_serve (%s): failed to bind socket (%s).\n", self->filename, strerror(errno));
139 }
140 else {
141 // En cas de succès, écouter les demandes de connexion.
142 listening = !listen(sock, 1);
143 if (!listening) {
144 fprintf(stderr, "LokiSocketServer_serve (%s): failed to start listening on socket (%s).\n", self->filename, strerror(errno));
145 }
146 }
147 }
148
149 while (!self->stop_request) {
150 pthread_mutex_lock(&self->mutex);
151 if (!self->has_data) {
152 pthread_cond_wait(&self->has_data_cond, &self->mutex);
153 }
154 if (self->stop_request) {
155 break;
156 }
157 const char *data = self->data;
158 self->has_data = false;
159 pthread_mutex_unlock(&self->mutex);
160
161 if (!listening) {
162 continue;
163 }
164
165 // Vérifier si une demande de connexion est présente en file d'attente.
166 // La socket est non bloquante.
167 if (accept_index < LIBLOKI_SOCKET_SERVER_CONNECTIONS) {
168 conn[accept_index] = accept(sock, NULL, NULL);
169 if (conn[accept_index] >= 0) {
170 fprintf(stderr, "LokiSocketServer_serve (%s): accepting connection %zu/%zu.\n", self->filename, accept_index, LIBLOKI_SOCKET_SERVER_CONNECTIONS);
171 // Marquer la connexion comme établie et chercher un slot libre
172 // pour la prochaine connexion.
173 status[accept_index] = true;
174 for (accept_index = 0; accept_index < LIBLOKI_SOCKET_SERVER_CONNECTIONS && status[accept_index]; accept_index ++);
175 }
176 }
177
178 // Pour chaque connexion ouverte, transmettre le paquet au client.
179 for (size_t i = 0; i < LIBLOKI_SOCKET_SERVER_CONNECTIONS; i ++) {
180 if (!status[i]) {
181 continue;
182 }
183
184 const char* remaining_data = data;
185 size_t remaining_size = self->data_size;
186 while (remaining_size > 0) {
187 ssize_t sz = write(conn[i], remaining_data, remaining_size);
188 if (sz >= 0) {
189 remaining_size -= sz;
190 remaining_data += sz;
191 }
192 else if (errno != EINTR) {
193 fprintf(stderr, "LokiSocketServer_serve (%s): write error on connection %zu (%s).\n", self->filename, i, strerror(errno));
194 status[i] = false;
195 accept_index = i;
196 break;
197 }
198 }
199 }
200 }
201
202 if (sock >= 0) {
203 close(sock);
204 }
205
206 if (has_file) {
207 unlink(self->filename);
208 }
209}

211int LokiSocketServer_start(LokiSocketServer *self, LokiFIFO *in, size_t cid) {
212 self->is_streaming = true;
213 self->in = in;
214 self->cid = cid;
215 self->stop_request = false;
216 int err = pthread_create(&self->receiver_thread, NULL, LokiSocketServer_receive, self);
217 if (err) {
218 return err;
219 }
220 return pthread_create(&self->server_thread, NULL, LokiSocketServer_serve, self);
221}

227void LokiSocketServer_stop(LokiSocketServer *self) {
228 if (self->stop_request) {
229 return;
230 }
231 self->stop_request = true;
232
233 // Débloquer le thread de réception.
234 LokiFIFO_abort(self->in);
235
236 // Débloquer le thread serveur.
237 pthread_mutex_lock(&self->mutex);
238 pthread_cond_signal(&self->has_data_cond);
239 pthread_mutex_unlock(&self->mutex);
240
241 // Attendre la terminaison normale des deux threads.
242 pthread_join(self->receiver_thread, NULL);
243 pthread_join(self->server_thread, NULL);
244}
